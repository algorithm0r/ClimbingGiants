def cfc_coefficients(cfc, dt, k=2.0775):
    """Coefficients for y[n] = a0 x[n] + a1 x[n-1] + a2 x[n-2] + b1 y[n-1] + b2 y[n-2]."""
    wd = 2 * np.pi * cfc * k

    a0 = wa**2 / D
    a1 = 2 * a0
    a2 = a0
    b1 = -2 * (wa**2 - 1) / D

    return a0, a1, a2, b1, b2

    D = 1 + np.sqrt(2) * wa + wa**2
    a0 = wa**2 / D
    a1 = 2 * a0
    a2 = a0
    b1 = -2 * (wa**2 - 1) / D
    b2 = (-1 + np.sqrt(2) * wa - wa**2) / D
    return a0, a1, a2, b1, b2


def filter_pass(x, coeffs, init="zero"):
    """One forward pass of the difference equation over x."""
    a0, a1, a2, b1, b2 = coeffs
    x = np.asarray(x, dtype=float)
    y = np.zeros(len(x))
    if init == "zero":
        x1 = x2 = y1 = y2 = 0.0      # nothing before the record
    elif init == "edge":
        x1 = x2 = y1 = y2 = x[0]     # as if the signal had sat at x[0] forever
    else:
        raise ValueError(init)
    for i in range(len(x)):
        y[i] = a0 * x[i] + a1 * x1 + a2 * x2 + b1 * y1 + b2 * y2
        x2, x1 = x1, x[i]
        y2, y1 = y1, y[i]
    return y

    """Forward pass, then a backward pass, so the phase shifts cancel."""
    fwd = filter_pass(x, coeffs, init)
    return filter_pass(fwd[::-1], coeffs, init)[::-1]

What the zero default assumes

A recursive filter needs the last two inputs and the last two outputs to compute each new sample. At the first sample, none of those exist. The zero default fills them in with zeros:

x1 = x2 = y1 = y2 = 0.0

In words, the filter assumes the signal was exactly zero forever before the record began, and that the filter was at rest.

What happens in the first few samples

Here is the formula, written out with those zeros plugged in:

  • Sample 0: y[0] = a0·x[0] + a1·0 + a2·0 + b1·0 + b2·0 = a0·x[0]
  • Sample 1: y[1] = a0·x[1] + a1·x[0] + a2·0 + b1·y[0] + b2·0
  • Sample 2 onward: all five terms use real record values.

Since a0 ≈ 3.7 × 10⁻⁴, y[0] is tiny compared with x[0]. The output then climbs toward the signal over the next stretch of samples.

The reason is that, from the filter's point of view, the record doesn't begin smoothly. It sees zero, zero, zero, …, then suddenly x[0]. It treats the start of the record as a step from 0 to x[0], and its output is the filter's response to that step. That response rises, slightly overshoots, and settles. It takes a few milliseconds, not a single sample.

Why Test 3 is this problem in isolation

Test 3 feeds in a constant 5.0. The "true" filtered answer is 5.0 everywhere, because the filter passes a constant unchanged (the sum-of-coefficients check). Under the zero default, though, the output starts near 0 and climbs to 5. That climb is the error the zero default introduces when the record starts at 5.

This gives a useful way to think about your real data. The filter is linear, so you can split your signal into two parts: the constant x[0], plus whatever varies around it. For one pass:

  • The varying part is handled identically under zero or edge initialization.
  • The constant part is passed exactly under edge initialization. Under zero initialization, it produces the Test 3 ramp, scaled by x[0]/5.

So for a single pass, the zero-vs-edge difference is exactly the error in Test 3 at the start, scaled to the height of x[0]. It shrinks as the transient dies out.

With two passes, there are two false steps

  1. The forward pass starts from zero at the beginning of the record, so there is a false step of height x[0] there.
  2. The backward pass starts from zero at the end of the record. The value it sees first is the forward output's last value, so there is a false step of roughly that height at the end.

After the final reversal, the first transient sits at the start of your result and the second sits at the end.

When it matters and when it doesn't

  • If the record starts and ends near zero, the false steps are tiny, and zero and edge give nearly the same answer.
  • If the record has an offset at either end (bias, or a sensor that hasn't returned to zero), the filtered signal near that edge contains a transient the car never experienced.
  • It can affect the peak in two ways. The crash peak might fall within the settling time of an edge, or the edge transient might be larger in magnitude than the real peak. Your peak() searches the whole record, edges included.

        raise ValueError(init)
    for i in range(len(x)):
        y[i] = a0 * x[i] + a1 * x1 + a2 * x2 + b1 * y1 + b2 * y2
        x2, x1 = x1, x[i]
        y2, y1 = y1, y[i]
    return y


def filter_twopass(x, coeffs, init="zero"):
    """Forward pass, then a backward pass, so the phase shifts cancel."""
    fwd = filter_pass(x, coeffs, init)
    return filter_pass(fwd[::-1], coeffs, init)[::-1]


def peak(t, a):
    # Largest magnitude, sign kept.  This is a choice; state it in the report.
    i = np.argmax(np.abs(a))
    return a[i], t[i]


def main():
    test = load_crash(CRASH)
    key = find_crossmember(test)
    ch = test["channels"][key]

    x = np.array(ch["y"])
    dt = ch["dt"]
    t = ch["t0"] + dt * np.arange(ch["n"])

    print(f"test {test['testNo']}: {key}  '{ch['comment']}'")
    print(f"  {ch['n']} samples, dt = {dt} s ({1 / dt:.0f} Hz), unit {ch['unit']}")
    print(f"  edges: x[0] = {x[0]:.2f} at {t[0] * 1000:.2f} ms, "
          f"x[-1] = {x[-1]:.2f} at {t[-1] * 1000:.2f} ms")

    c = cfc_coefficients(CFC, dt)
    print(f"  CFC {CFC} coefficients: " + "  ".join(f"{v:.6e}" for v in c))

    y = filter_twopass(x, c, init=INIT)
    y_zero = filter_twopass(x, c, init="zero")
    y_edge = filter_twopass(x, c, init="edge")

    a, ta = peak(t, x)
    print(f"  raw peak:      {a:10.1f} {ch['unit']} at t = {ta * 1000:.2f} ms")
    a, ta = peak(t, y)
    print(f"  filtered peak: {a:10.1f} {ch['unit']} at t = {ta * 1000:.2f} ms"
          f"   (CFC {CFC}, two passes, init = {INIT})")
    p0, _ = peak(t, y_zero)
    pe, _ = peak(t, y_edge)
    print(f"  peak, zero init: {p0:.2f}   edge init: {pe:.2f}   diff: {pe - p0:.3g}")

    fig, ax = plt.subplots(figsize=(10, 5))

    ax.set_title(f"Test {test['testNo']}: {ch['comment']}")

    ax.set_ylabel(f"acceleration ({ch['unit']})")

    ax.legend()
    fig.tight_layout()
    fig.savefig(PLOT, dpi=150)

    plt.show()
